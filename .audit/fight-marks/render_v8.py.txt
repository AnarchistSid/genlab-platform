One anchor rule for every shot is why the water-breathing sweep rendered as a
sliver: a 9:16 crop of a 16:9 frame is 31.6% of its width, and the sweep's
arc spans 58-100%. Character shots still anchor on the face. ACTION shots
anchor on the focal point -- and when the action will not fit inside the crop
with a 10% margin, the shot is shown WIDE on a clean matte for a beat rather
than cropped into a fragment.
"""

# a full-height 1:1 crop is 56.2% of a 16:9 frame's width and 4:5 is 45.0%, so

PULSE_MAX = 0.03
PULSE_DECAY_S = 0.125           # the pulse is a kick at the cut, then nothing


def run(cmd, t=2400):
    r = subprocess.run(cmd, capture_output=True, text=True, timeout=t)
    if r.returncode:
        raise RuntimeError(f"{cmd[0]}: {r.stderr[-1500:]}")
    return r


def mask_filter(fight_id: str, src_w: int = 1920, src_h: int = 1080) -> str:
    """Remove every burnt-in mark BEFORE the crop magnifies it.

    §3 asks for blur + darken. `delogo` is a better realisation of the same
    intent and was chosen after comparing all three on the same frame: a
    black fill is an obvious rectangle, blur+darken is a visible band with a
    hard edge, and delogo reconstructs from the surrounding pixels so the
    mark simply is not there. A mark magnified is a mark featured -- but so
    is a black box where one used to be.
    """
    cfg = yaml.safe_load(open("genlab-core/config/source_watermarks.yaml"))
    boxes = cfg["channels"][cfg["fights"][fight_id]]
    out = []
    for b in boxes:
        x0, y0, x1, y1 = b["box"]
        x = max(1, int(x0 * src_w)); y = max(1, int(y0 * src_h))
        w = min(int((x1 - x0) * src_w), src_w - x - 1)
        h = min(int((y1 - y0) * src_h), src_h - y - 1)
        if w > 4 and h > 4:
            out.append(f"delogo=x={x}:y={y}:w={w}:h={h}")
    return ",".join(out) if out else "null"


def wide_filter(focal, src_w, src_h, aspect=16 / 9, pulse=0.0, pulse_period=0.8,
                shake_at=()):
    """The whole frame, 1080 wide, on a clean dark matte.

    Not a blurred band: a blurred copy of the picture behind the picture is
    the amateur tell. A flat dark matte says "this is a wide beat" instead of
    pretending the frame is taller than it is. The focal point is centred
    vertically so the action sits where the eye already is.
    """
    # crop the source to the chosen aspect (full height), then letterbox it
    cw = int(round(min(src_w, src_h * aspect))); cw -= cw % 2
    fx = (focal[0] if focal else 0.5)
    cx = int(round(min(max(0.0, fx * src_w - cw / 2), src_w - cw)))
    ch = int(round(OUT_W * src_h / cw)); ch -= ch % 2
    fy = (focal[1] if focal else 0.5)
    top = max(0, min(int(round(OUT_H / 2 - fy * ch)), max(0, OUT_H - ch)))
    f = [f"crop={cw}:{src_h}:{cx}:0", f"scale={OUT_W}:{ch}:flags=lanczos"]
    if ch < OUT_H:
        f.append(f"pad={OUT_W}:{OUT_H}:0:{top}:color={MATTE}ff")
    else:
        f.append(f"crop={OUT_W}:{OUT_H}")
    f.append(_camera(pulse, pulse_period, shake_at))
    f.append("setsar=1")
    return ",".join(f)


def _camera(pulse: float, pulse_period: float, shake_at=()) -> str:
    """The camera. STILL unless a hit says otherwise.

    Continuous shake reads as a nervous camera, not as energy, and it makes
    slow motion look like a handheld phone. Shake is a hit: a few frames,
    then still. The pulse is on downbeats only and capped at 3%.
    """
    parts = []
    if shake_at:
        dur = SHAKE_FRAMES / FPS
        conds = "+".join(f"between(t,{a:.3f},{a + dur:.3f})" for a in shake_at)
        parts.append(
            f"crop=w=iw-{int(SHAKE_PX * 2)}:h=ih-{int(SHAKE_PX * 2)}"
            f":x='{SHAKE_PX}+{SHAKE_PX:.1f}*({conds})*sin(60*t)'"
            f":y='{SHAKE_PX}+{SHAKE_PX:.1f}*({conds})*cos(57*t)',"
            f"scale={OUT_W}:{OUT_H}")
    if pulse:
        # A boxcar at the cut, not a continuous oscillation. A sine over the
        # whole shot is a camera that never stops moving: measured, it put the
        # reel at 3.15 px of global motion against a 0.84 px source floor and
        # a 1.5 px cap. The cut lands on the grid, so a kick there IS the
        # downbeat.
        z = (f"1+{min(pulse, PULSE_MAX):.4f}*"
             f"(1-t/{PULSE_DECAY_S:.3f})*between(t,0,{PULSE_DECAY_S:.3f})")
        parts.append(f"scale=w='iw*({z})':h='ih*({z})':eval=frame,crop={OUT_W}:{OUT_H}")
    return ",".join(parts) if parts else "null"


def crop_filter(anchor, src_w, src_h, pulse=0.0, pulse_period=0.8,
                split_at=(), shake_at=()):
    """Full-screen 9:16 on the subject's face box, with the genre's movement."""
    full_bleed = int(src_h * OUT_W // OUT_H)
    cx = (anchor[0] if anchor else 0.5) * src_w
    cw = min(full_bleed, src_w)
    x = int(round(min(max(0.0, cx - cw / 2), src_w - cw)))
    f = [f"crop={cw}:{src_h}:{x}:0",
         f"scale={OUT_W}:{OUT_H}:force_original_aspect_ratio=increase:flags=lanczos",
         f"crop={OUT_W}:{OUT_H}",
         _camera(pulse, pulse_period, shake_at)]
    if split_at:
        cond = "+".join(f"between(t,{a:.3f},{a + 2 / FPS:.3f})" for a in split_at)
        f.append(f"rgbashift=rh=2:bh=-2:enable='{cond}'")
    f.append("setsar=1")
    return ",".join(x for x in f if x != "null")


MAX_ON_SCREEN_S = 2.5


def split_for_screen_time(segs):
    """Cap what is ON SCREEN, not what was in the source.

    A 2.5 s source span played at 0.6x is 4.17 s of one composition. The cap
    is about the hold a viewer sees, so it is applied after the speed is
    known -- and the split gives each piece its own crop, which makes it a
    new composition rather than a longer one.
    """
    out = []
    for s in segs:
        speed = SPEED.get(s["shot_type"], 1.0) if s["kind"] != "hit" else 0.25
        on_screen = (s["end"] - s["start"]) / max(speed, 1e-6)
        n = max(1, int((on_screen - 1e-6) // MAX_ON_SCREEN_S) + 1)
        if n == 1 or s["kind"] == "hit":
            out.append(s); continue
        step = (s["end"] - s["start"]) / n
        for k in range(n):
            out.append({**s, "start": round(s["start"] + k * step, 3),
                        "end": round(s["start"] + (k + 1) * step, 3)})
    return out


def load_frames():
    d = json.loads((SP / "vision" / "frames_v7.json").read_text())["frames"]
    return {round(f["t"], 2): f for f in d}


def frame_near(frames, t):
    return frames[min(frames, key=lambda k: abs(k - t))]


def aspect_for(action_w: float, src_w: int = 1920, src_h: int = 1080):
    """The FIRST aspect whose full-height crop holds the action with margin."""
    for name, ar in ASPECTS:
        w = min(src_w, src_h * ar)
        if action_w * ACTION_MARGIN <= w / src_w:
            return name, ar, w / src_w
    name, ar = ASPECTS[-1]
    return name, ar, 1.0


def decide_layout(frames, shot, src_w=1920, src_h=1080):
    """Face anchor for character shots; for action, the tightest aspect that holds it."""
    f = frame_near(frames, (shot["start"] + shot["end"]) / 2)
    ty = shot["shot_type"]
    fp, ae = f.get("focal_point"), f.get("action_extent")
    if ty not in ("clash", "impact"):
        return {"layout": "crop-face", "aspect": "9:16", "anchor": shot.get("anchor"),
                "focal": fp, "action_w": (ae[2] - ae[0]) if ae else None}
    aw = (ae[2] - ae[0]) if ae else 1.0
    name, ar, frac = aspect_for(aw, src_w, src_h)
    if name == "9:16":
        return {"layout": "crop-focal", "aspect": name, "anchor": [fp[0], fp[1]] if fp else None,
                "focal": fp, "action_w": aw}
    return {"layout": f"wide-{name}", "aspect": name, "anchor": None,
            "focal": fp or [0.5, 0.5], "action_w": aw}


def cut_long_holds(frames, segs):
    """A drawing held >= HOLD_CAP_S cuts, however much pulses over it.

    The `hold` field marks a frame as the same DRAWING as the one before it.
    A run of them is a freeze wearing a shake.
    """
    out = []
    for s in segs:
        ts = sorted(t for t in frames if s["start"] <= t < s["end"])
        run_start, cut_at = None, []
        for t in ts:
            if frames[t].get("hold"):
                run_start = run_start if run_start is not None else t
                if t - run_start >= HOLD_CAP_S:
                    cut_at.append(t); run_start = None
            else:
                run_start = None
        if not cut_at:
            out.append(s); continue
        a = s["start"]
        for c in cut_at + [s["end"]]:
            if c - a > 0.2:
                out.append({**s, "start": round(a, 3), "end": round(c, 3),
                            "held_cut": c != s["end"]})
            a = c
    return out


def segments_from(shots):
    """Split any shot that straddles the hit; the hit is its own segment."""
    segs = []
    for s in shots:
        a, b, ty = s["start"], s["end"], s["shot_type"]
        if b <= HIT_IN or a >= HIT_OUT:
            segs.append({**s, "kind": "plain"})
            continue
        if a < HIT_IN:
            segs.append({**s, "end": HIT_IN, "kind": "plain"})
        segs.append({**s, "start": max(a, HIT_IN), "end": min(b, HIT_OUT), "kind": "hit"})
        if b > HIT_OUT:
            segs.append({**s, "start": HIT_OUT, "kind": "plain"})
    return split_for_screen_time([s for s in segs if s["end"] - s["start"] > 0.03])


def build(fight_id="tanjiro_vs_akaza"):
    kit = load_kit("anime"); brand = TY.for_niche("anime")
    work = SP / "v8"; work.mkdir(parents=True, exist_ok=True)
    src = SP / "fights" / f"{fight_id}.mp4"
    sl = json.loads((SP / "vision" / "shotlist_v6.json").read_text())
    music = json.loads((SP / "music_v4_final.json").read_text())[fight_id]
    # The hit is slowed by Topaz, so its FRAMING is baked in before the
    # interpolation runs. A hit whose action does not fit a 9:16 crop has to
    # be sent wide -- otherwise the layout decision says "wide" and the file
    # on disk is the cropped one.
    hit_wide = SP / "interp" / "hit_wide_slow_trim.mp4"   # still camera: no pulse, no shake
    hit_crop = SP / "interp" / "hit_slow.mp4"
    mask = mask_filter(fight_id)
    w, h = 1920, 1080
    log = {"fight": fight_id, "fps": FPS, "impact_src": IMPACT, "segments": [],
           "cost_usd": 0.0, "shotlist": "vlm+operator_corrections"}
    frames = load_frames()
    segs = cut_long_holds(frames, segments_from(sl["shots"]))
    for s in segs:
        s.update(decide_layout(frames, s))
        if s["layout"].startswith("wide-"):
            d = s["end"] - s["start"]
            if d > WIDE_MAX_S:
                s["end"] = round(s["start"] + WIDE_MAX_S, 3)
    # §3 — shots at 1:1 and wider may not exceed WIDE_BUDGET of the reel. The
    # widest are shortened first, to their best 1.0 s, because a wide beat is
    # a beat and not a section.
    order = {n: i for i, (n, _) in enumerate(ASPECTS)}
    wide_i = order[WIDE_FROM]
    total = sum(x["end"] - x["start"] for x in segs) or 1.0
    wide = [x for x in segs if order.get(x.get("aspect"), 0) >= wide_i]
    wide.sort(key=lambda x: -order.get(x.get("aspect"), 0))
    for x in wide:
        share = sum(y["end"] - y["start"] for y in wide) / total
        if share <= WIDE_BUDGET:
            break
        if x["end"] - x["start"] > WIDE_MIN_S:
            x["end"] = round(x["start"] + WIDE_MIN_S, 3)
            x["wide_trimmed"] = True
    parts, reel_t, impact_rel = [], 0.0, None
    for i, s in enumerate(segs):
        a, b = s["start"], s["end"]
        p = work / f"seg_{i:02d}.mp4"
        if s["kind"] == "hit":
            # Topaz already slowed it; it is the only segment not cut from source.
            # startswith, not an equality on a layout NAME: the names became
            # wide-{aspect} in PEAK-15 and an == "wide-4:5" test stopped
            # matching silently, falling back to the cropped hit.
            wide_layout = str(s.get("layout", "")).startswith("wide-")
            hit_clip = hit_wide if (wide_layout and hit_wide.exists()) else hit_crop
            log.setdefault("hit_source", hit_clip.name)
            run(["ffmpeg", "-y", "-v", "error", "-i", str(hit_clip), "-an",
                 "-vf", f"fps={FPS}", "-r", str(FPS), *X264, str(p)])
            impact_rel = reel_t + (IMPACT - a) * (probe_duration(p) / max(b - a, 1e-6))
            speed, interp = round((b - a) / probe_duration(p), 3), True
        else:
            speed = SPEED.get(s["shot_type"], 1.0)
            interp = False
            # §1 — the camera is still. Pulse only where the shot is on the
            # grid and moving; shake only on a hit shot, never on a hold.
            pulse = 0.02 if (speed >= 1 and not s.get("held_cut")) else 0.0
            shake = (0.0,) if s["shot_type"] == "impact" else ()
            ar = dict(ASPECTS).get(s.get("aspect"), 16 / 9)
            if s.get("layout", "").startswith("wide-"):
                inner = wide_filter(s.get("focal"), w, h, aspect=ar,
                                    pulse=pulse, shake_at=shake)
            else:
                inner = crop_filter(s.get("anchor"), w, h, pulse=pulse,
                                    split_at=(0.0,) if speed >= 1 else (),
                                    shake_at=shake)
            vf = f"{mask},{inner},setpts={1 / speed:.4f}*PTS,fps={FPS}"
            run(["ffmpeg", "-y", "-v", "error", "-ss", f"{a:.3f}", "-t", f"{b - a:.3f}",
                 "-i", str(src), "-an", "-vf", vf, *X264, str(p)])
        d = probe_duration(p)
        log["segments"].append({"i": i, "src": [round(a, 3), round(b, 3)],
                                "subject": s.get("subject"), "type": s["shot_type"],
                                "kind": s["kind"], "speed": speed, "interpolated": interp,
                                "reel_start": round(reel_t, 3), "out_s": round(d, 3),
                                "anchor": s.get("anchor"), "layout": s.get("layout"),
                                "aspect": s.get("aspect"),
                                "wide_trimmed": s.get("wide_trimmed", False),
                                "focal": s.get("focal"), "action_w": s.get("action_w"),
                                "held_cut": s.get("held_cut", False)})
        parts.append(p); reel_t += d
    # §3 — the model checks the RENDERED action shots before anyone sees them.
    from genlab_core.intel import frame_vision as FV
    FV.VISION_APP = "google/gemini-3-6-flash"
    shots_to_check = [(i, sg) for i, sg in enumerate(log["segments"])
                      if sg["type"] in ("clash", "impact")]
    imgs, labels, owner = [], [], []
    for i, sg in shots_to_check:
        src_clip = parts[sg["i"]]
        for frac in (0.35, 0.7):
            q = work / f"chk_{sg['i']:02d}_{int(frac * 100)}.png"
            # §4 — at PHONE size. A frame that reads at 512 px can be
            # illegible at the size it is actually watched.
            run(["ffmpeg", "-y", "-v", "error", "-ss", f"{sg['out_s'] * frac:.3f}",
                 "-i", str(src_clip), "-frames:v", "1", "-vf", "scale=360:-2", str(q)])
            imgs.append(q); owner.append(sg["i"])
            labels.append(f"{sg['type']} shot, layout {sg.get('layout')}")
    verdicts = FV.check_framing(imgs, labels) if imgs else []
    per_shot = {}
    for v, o in zip(verdicts, owner):
        per_shot.setdefault(o, []).append(v)
    # A description that names no action is a fail, however confident it is.
    seg_by_i = {sg["i"]: sg for sg in log["segments"]}
    fc = {}
    for k, vs in per_shot.items():
        sg = seg_by_i[k]
        terms = tuple(x for x in (sg.get("subject"), sg.get("type"), "Tanjiro",
                                  "Akaza", "Giyu") if x)
        named = [FV.describes_action(v.get("describe", ""), terms) for v in vs]
        legible = [v.get("legible_at_this_size") for v in vs]
        fc[str(k)] = {"describe": [v.get("describe") for v in vs],
                      "names_action": named, "legible": legible,
                      "pass": all(named) and all(x is True for x in legible)}
    log["framing_check"] = fc
    log["framing_failed"] = [int(k) for k, v in fc.items() if not v["pass"]]

    # §4 — a fail flips the layout one step wider. At the widest aspect there
    # is no wider, and the remaining lever is DURATION: when the early frame
    # reads and the late one does not, the shot is trimmed to the part that
    # reads. Measured on the Topaz hit, whose tail is a blurred blue flash the
    # model cannot follow at phone size.
    trims = {}
    for k, v in fc.items():
        if v["pass"]:
            continue
        early_ok = v["names_action"][0] and v["legible"][0] is True
        late_ok = len(v["legible"]) > 1 and v["legible"][1] is True
        if early_ok and not late_ok:
            trims[int(k)] = 0.60
    log["framing_trims"] = trims
    log["body_s"] = round(reel_t, 2)
    log["impact_rel"] = round(impact_rel, 3) if impact_rel else None
    log["remapped"] = sum(1 for x in log["segments"] if x["speed"] != 1.0)
    log["interpolated"] = sum(1 for x in log["segments"] if x["interpolated"])
    return log, parts, work, kit, brand, src, music


def finish(log, parts, work, kit, brand, src, music, out):
    from genlab_core.still.sound import generate_sfx

    lst = work / "concat.txt"
    lst.write_text("".join(f"file '{p.resolve()}'\n" for p in parts))
    body = work / "body.mp4"
    run(["ffmpeg", "-y", "-v", "error", "-f", "concat", "-safe", "0", "-i", str(lst),
         "-r", str(FPS), *X264, str(body)])
    vlen = probe_duration(body)
    impact_rel = log["impact_rel"] or vlen / 2
    bpm = float(music["pick_bpm"])

    bed = Path(music["pick"]); aligned = work / "bed.m4a"
    log["bed_fit"] = MU.fit_bed(bed, float(music["pick_drop"]), impact_rel, vlen, bpm, aligned)
    log["bed"] = {"register": music["register"], "bpm": bpm, "drop": music["pick_drop"]}

    hit = generate_sfx("sub_drop", work / "hit.mp3", duration_s=2.0)
    log["cost_usd"] += 0.00104
    src_a = work / "src_audio.m4a"
    run(["ffmpeg", "-y", "-v", "error", "-ss", f"{log['segments'][0]['src'][0]:.3f}",
         "-i", str(src.with_suffix(".m4a")), "-vn",
         "-af", f"apad,atrim=0:{vlen:.3f}", "-c:a", "aac", "-b:a", "192k",
         "-ar", "48000", "-ac", "2", str(src_a)])
    premix = work / "premix.m4a"
    run(["ffmpeg", "-y", "-v", "error", "-i", str(src_a), "-i", str(aligned), "-i", str(hit),
         "-filter_complex",
         f"[0:a]volume=-8dB,apad,atrim=0:{vlen:.3f},asetpts=PTS-STARTPTS[show];"
         f"[2:a]volume=0dB,afade=t=out:st=0.8:d=1.2,"
         f"adelay={int(impact_rel * 1000)}|{int(impact_rel * 1000)},"
         f"apad,atrim=0:{vlen:.3f},asetpts=PTS-STARTPTS[hitm];"
         f"[2:a]volume=0dB,afade=t=out:st=0.8:d=1.2,"
         f"adelay={int(impact_rel * 1000)}|{int(impact_rel * 1000)},"
         f"apad,atrim=0:{vlen:.3f},asetpts=PTS-STARTPTS[key];"
         f"[1:a]volume=-7dB[bv];[key]asplit=2[k1][k2];"
         f"[bv][k1]sidechaincompress=threshold=0.03:ratio=8:attack=5:release=150[bed];"
         f"[show][k2]sidechaincompress=threshold=0.05:ratio=4:attack=5:release=200[sh];"
         f"[sh][bed][hitm]amix=inputs=3:duration=first:normalize=0,"
         f"apad,atrim=0:{vlen:.3f}[a]", "-map", "[a]", "-c:a", "aac", "-b:a", "192k",
         "-ar", "48000", "-ac", "2", str(premix)])
    target = float(kit["audio"]["target_lufs"]) + 0.14
    cur, passes = premix, []
    for _ in range(4):
        i, tp = measure_loudness(cur)
        g = target - i
        passes.append({"lufs": round(i, 2), "tp": round(tp, 2), "gain": round(g, 2)})
        if abs(g) < 0.15:
            break
        nxt = work / "n_tmp.m4a"
        run(["ffmpeg", "-y", "-v", "error", "-i", str(cur), "-af",
             f"volume={g:.2f}dB,alimiter=limit=-1.5dB:level=disabled",
             "-c:a", "aac", "-b:a", "192k", "-ar", "48000", "-ac", "2", str(nxt)])

    log["normalise"] = passes
